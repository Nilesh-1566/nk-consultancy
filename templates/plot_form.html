// एकापेक्षा जास्त फोटो (Front + Back + PAN) प्रोसेस करणे async function handleFiles(input, boxId) { if (!input.files || input.files.length === 0) return; const files = Array.from(input.files); const status = document.getElementById('status_' + boxId); const box = document.getElementById(boxId); status.innerText = `स्कॅन सुरू आहे... (${files.length} फोटो)`; let combinedText = ''; for (let i = 0; i < files.length; i++) { const file = files[i]; status.innerText = `फोटो ${i + 1} वाचत आहे... कृपया थांबा.`; try { let imageBlob = file; if (file.type === 'application/pdf') { const arrayBuffer = await file.arrayBuffer(); const pdf = await pdfjsLib.getDocument({data: arrayBuffer}).promise; const page = await pdf.getPage(1); const viewport = page.getViewport({scale: 1.5}); const canvas = document.createElement('canvas'); canvas.height = viewport.height; canvas.width = viewport.width; await page.render({canvasContext: canvas.getContext('2d'), viewport: viewport}).promise; imageBlob = await new Promise(resolve => canvas.toBlob(resolve, 'image/jpeg')); } // मराठी (mar) + इंग्रजी (eng) दोन्ही भाषा ओळखणे const res = await Tesseract.recognize(imageBlob, 'mar+eng', { logger: m => { if (m.status === 'recognizing text') { status.innerText = `मराठी मजकूर वाचत आहे: ${Math.round(m.progress * 100)}%`; } } }); combinedText += '\n' + res.data.text; } catch (e) { console.error('File OCR error:', e); } } status.innerText = '✅ माहिती वाचण्यात आली! मराठी मजकूर भरत आहे...'; parseAndFillData(combinedText, box); } // गोळा झालेल्या मजकुरातून मराठी नाव, मराठी पत्ता, वय, पॅन आणि आधार शोधणे function parseAndFillData(text, box) { const lines = text.split('\n').map(l => l.trim()).filter(l => l.length > 0); // देवनागरी (मराठी) अक्षरे तपासण्यासाठी रेग्युलर एक्स्प्रेशन const marathiRegex = /[\u0900-\u097F]/; // १. PAN क्रमांक शोधणे (इंग्रजी) const panMatch = text.match(/[A-Z]{5}[0-9]{4}[A-Z]{1}/); if (panMatch && !box.querySelector('.field_pan').value) { box.querySelector('.field_pan').value = panMatch[0]; } // २. आधार क्रमांक शोधणे (४-४-४ अंक) const aadharMatch = text.match(/\b\d{4}\s?\d{4}\s?\d{4}\b/); if (aadharMatch && !box.querySelector('.field_aadhar').value) { box.querySelector('.field_aadhar').value = aadharMatch[0]; } // ३. मोबाईल नंबर शोधणे const mobMatch = text.match(/\b[6-9]\d{9}\b/); if (mobMatch && !box.querySelector('.field_mobile').value) { box.querySelector('.field_mobile').value = mobMatch[0]; } // ४. वय / जन्मतारीख शोधणे const currentYear = 2026; const dobMatch = text.match(/(?:DOB|Birth|जन्म|तारीख|वर्ष)[:\s/]*([0-9]{4}|[0-9]{2}[/-][0-9]{2}[/-][0-9]{4})/i); if (dobMatch) { let yearStr = dobMatch[1]; if (yearStr.length > 4) yearStr = yearStr.slice(-4); const y = parseInt(yearStr); if (y > 1920 && y <= currentYear) { box.querySelector('.field_age').value = `वय ${currentYear - y} वर्ष`; } } // ५. मराठी पत्ता शोधणे (आधारच्या मागच्या बाजूवरील "पत्ता:" किंवा "मु. पो.") let marathiAddressLines = []; let addressStart = false; for (let i = 0; i < lines.length; i++) { const line = lines[i]; // पत्ता सुरू होण्याची खूण if (/^(पत्ता|मु\.?\s*पो|रा\.|Address)/i.test(line) || line.includes('पत्ता:')) { addressStart = true; // "पत्ता:" हा शब्द काढून फक्त पत्ता घेणे const cleanLine = line.replace(/^(पत्ता|Address)[:\s]*/gi, ''); if (cleanLine.length > 0) marathiAddressLines.push(cleanLine); continue; } // पत्ता सुरू झाल्यानंतरच्या मराठी ओळी गोळा करणे if (addressStart) { // जर पिनकोड आला किंवा इंग्रजी हेडर आले तर पत्ता संपला if (line.match(/\b(4\d{5})\b/) || marathiAddressLines.length >= 4) { marathiAddressLines.push(line); break; } if (marathiRegex.test(line)) { marathiAddressLines.push(line); } } } // पिन कोड शोधणे const pinMatch = text.match(/\b(4\d{5})\b/); if (marathiAddressLines.length > 0) { let fullAddr = marathiAddressLines.join(', ').replace(/\s+/g, ' '); if (pinMatch && !fullAddr.includes(pinMatch[0])) { fullAddr += ', पिन कोड- ' + pinMatch[0]; } box.querySelector('.field_address').value = fullAddr; } else if (pinMatch && !box.querySelector('.field_address').value) { box.querySelector('.field_address').value = 'पिन कोड- ' + pinMatch[0]; } // ६. मराठी नाव शोधणे (फक्त शुद्ध देवनागरी/मराठी अक्षरे असलेली ओळ) if (!box.querySelector('.field_name').value) { for (let i = 0; i < lines.length; i++) { const line = lines[i]; // सरकारी मथळे व शब्द वगळणे if (/भारत|सरकार|प्राधिकरण|शासकीय|नोंदणी|जन्म|तारीख|वय|पुरुष|स्त्री|पत्ता|मु\.?\s*पो/i.test(line)) { continue; } // ओळीत मराठी अक्षरे असावीत आणि इंग्रजी किंवा अंक नसावेत if (marathiRegex.test(line) && !/[a-zA-Z0-9]/.test(line)) { const words = line.split(/\s+/).filter(w => w.length > 1); // व्यक्तीचे नाव सहसा २ ते ४ शब्दांचे असते if (words.length >= 2 && words.length <= 4 && line.length > 6) { box.querySelector('.field_name').value = line.trim(); break; } } } } }
