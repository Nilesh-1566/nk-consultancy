// गोळा झालेल्या सर्व मजकुरातून मराठी नाव, पत्ता, वय, पॅन आणि आधार शोधणारे प्रगत लॉजिक function parseAndFillData(text, box) { // सर्व ओळी स्वच्छ करणे const rawLines = text.split('\n').map(l => l.trim()).filter(l => l.length > 0); // देवनागरी (मराठी) अक्षरे तपासण्यासाठी const marathiRegex = /[\u0900-\u097F]/; // १. PAN क्रमांक शोधणे (१० अंकी कॅपिटल) const panMatch = text.match(/[A-Z]{5}[0-9]{4}[A-Z]{1}/); if (panMatch && !box.querySelector('.field_pan').value) { box.querySelector('.field_pan').value = panMatch[0]; } // २. आधार क्रमांक शोधणे (४-४-४ अंक) const aadharMatch = text.match(/\b\d{4}\s?\d{4}\s?\d{4}\b/); if (aadharMatch && !box.querySelector('.field_aadhar').value) { box.querySelector('.field_aadhar').value = aadharMatch[0]; } // ३. मोबाईल नंबर शोधणे const mobMatch = text.match(/\b[6-9]\d{9}\b/); if (mobMatch && !box.querySelector('.field_mobile').value) { box.querySelector('.field_mobile').value = mobMatch[0]; } // ४. वय / जन्मतारीख (DOB / Year of Birth) शोधणे const currentYear = 2026; const dobMatch = text.match(/(?:DOB|Birth|जन्म|तारीख|वर्ष)[:\s/]*([0-9]{4}|[0-9]{2}[/-][0-9]{2}[/-][0-9]{4})/i); if (dobMatch) { let yearStr = dobMatch[1]; if (yearStr.length > 4) yearStr = yearStr.slice(-4); const y = parseInt(yearStr); if (y > 1920 && y <= currentYear) { box.querySelector('.field_age').value = `वय ${currentYear - y} वर्ष`; } } // ५. मराठी पत्ता शोधणे (Address Extraction) const pinMatch = text.match(/\b(4\d{5})\b/); let addressLines = []; let foundAddressIndex = -1; // आधार कार्डच्या मागच्या बाजूचा पत्ता सुरू होणारे संभाव्य शब्द for (let i = 0; i < rawLines.length; i++) { const line = rawLines[i]; if (/^(पत्ता|पता|प्ता|Address|मु\.?\s*पो|C\/O|S\/O|W\/O|D\/O|आत्मज|पत्नी)/i.test(line) || line.includes('पत्ता:') || line.includes('Address:')) { foundAddressIndex = i; break; } } // पत्ता सुरू झाल्यावर पुढील ३ ते ५ ओळी गोळा करणे (पिनकोड येईपर्यंत) if (foundAddressIndex !== -1) { for (let j = foundAddressIndex; j < Math.min(foundAddressIndex + 6, rawLines.length); j++) { let l = rawLines[j]; // "पत्ता:" किंवा "Address:" हा शब्द काढून टाकणे l = l.replace(/^(पत्ता|पता|प्ता|Address|C\/O|S\/O)[:\s\-]*/gi, '').trim(); // आधार नंबर किंवा इंग्रजी QR कोड कचरा वगळणे if (/Unique|Authority|www|uidai|\d{4}\s?\d{4}/i.test(l)) continue; if (l.length > 2) { addressLines.push(l); } // पिनकोड सापडल्यास पत्ता तिथेच पूर्ण झाला if (/\b(4\d{5})\b/.test(l)) { break; } } } if (addressLines.length > 0) { let completeAddress = addressLines.join(', ').replace(/\s+/g, ' '); // शेवटी पिनकोड व्यवस्थित जोडणे if (pinMatch && !completeAddress.includes(pinMatch[0])) { completeAddress += ', पिन कोड- ' + pinMatch[0]; } box.querySelector('.field_address').value = completeAddress; } else if (pinMatch && !box.querySelector('.field_address').value) { box.querySelector('.field_address').value = 'पिन कोड- ' + pinMatch[0]; } // ६. मराठी संपूर्ण नाव शोधणे (Name Extraction) if (!box.querySelector('.field_name').value) { for (let i = 0; i < rawLines.length; i++) { const line = rawLines[i]; // सरकारी मथळे व अनावश्यक शब्द वगळणे if (/भारत|सरकार|Government|India|Unique|Identification|Authority|प्राधिकरण|शासकीय|नोंदणी|जन्म|तारीख|DOB|Year|Male|Female|पुरुष|स्त्री|पत्ता|Address|मु\.?\s*पो/i.test(line)) { continue; } // ओळीत मराठी (देवनागरी) अक्षरे असली पाहिजेत आणि अंक नसावेत if (marathiRegex.test(line) && !/[0-9]/.test(line)) { const words = line.split(/\s+/).filter(w => w.length > 1); // व्यक्तीचे नाव २ ते ४ शब्दांचे असते आणि लांबी ६ अक्षरांपेक्षा जास्त असावी if (words.length >= 2 && words.length <= 4 && line.length >= 6) { box.querySelector('.field_name').value = line.trim(); break; } } } } }
